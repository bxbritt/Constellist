};

// Add links to the pane (glow layer first, then core)
for (StarLink link : links) {
getChildren().addAll(link.getGlowLine(), link.getCoreLine());
}

// Add stars on top of links
getChildren().addAll(stars);

// Draw all links
for (StarLink link : links) {
link.draw();
}
